cmake_minimum_required(VERSION 3.20)
project(receipt_core VERSION 1.0.0 LANGUAGES C CXX)

option(RECEIPT_CORE_SANITIZERS "Enable host AddressSanitizer and UBSan" OFF)
add_library(receipt_core receipt_core.cpp)
target_include_directories(receipt_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(receipt_core PUBLIC cxx_std_11)
set_target_properties(receipt_core PROPERTIES CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE YES)
target_compile_definitions(receipt_core PRIVATE RECEIPT_CORE_BUILD)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(receipt_core PUBLIC RECEIPT_CORE_SHARED)
endif()
if(MSVC)
  target_compile_options(receipt_core PRIVATE /W4 /WX /EHs-c-)
else()
  target_compile_options(receipt_core PRIVATE -Wall -Wextra -Werror -fno-exceptions)
endif()
if(RECEIPT_CORE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers require Clang or GCC")
  endif()
  target_compile_options(receipt_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(receipt_core PUBLIC -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(NODE_EXECUTABLE NAMES node REQUIRED)
  add_executable(receipt_core_test tests/core_test.cpp)
  target_link_libraries(receipt_core_test PRIVATE receipt_core Threads::Threads)
  add_test(NAME core COMMAND receipt_core_test)
  set_tests_properties(core PROPERTIES TIMEOUT 10)
  add_executable(receipt_core_c_test tests/c_header_test.c)
  set_target_properties(receipt_core_c_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
  target_link_libraries(receipt_core_c_test PRIVATE receipt_core)
  add_test(NAME c_abi COMMAND receipt_core_c_test)
  add_executable(receipt_core_corpus tests/corpus_driver.cpp)
  target_link_libraries(receipt_core_corpus PRIVATE receipt_core)
  add_test(NAME shared_corpus COMMAND "${NODE_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/corpus_runner.js" "$<TARGET_FILE:receipt_core_corpus>")
  add_test(NAME vendor_snapshot COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/vendor_test.py")
endif()
