cmake_minimum_required(VERSION 3.20)
project(receipt_overlap_transport_tests LANGUAGES C CXX)
include(CTest)
find_package(Threads REQUIRED)
add_subdirectory(../../receipt_core receipt_core)
add_executable(receipt_transport_test transport_test.cpp)
target_compile_features(receipt_transport_test PRIVATE cxx_std_20)
target_compile_options(receipt_transport_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(receipt_transport_test PRIVATE receipt_core Threads::Threads)
add_test(NAME numeric_transport COMMAND receipt_transport_test)
add_executable(receipt_transport_driver transport_driver.cpp)
target_compile_features(receipt_transport_driver PRIVATE cxx_std_20)
target_link_libraries(receipt_transport_driver PRIVATE receipt_core)
